# Production Configuration — api-gateway (prod-us-east-1)
# WARNING: This file is READ ONLY. Do not modify without change control.

server {
    listen 443 ssl http2;
    server_name api.company.com;

    # Upstream services
    upstream user-service {
        server user-service:8081 max_fails=3 fail_timeout=30s;
        keepalive 200;  # Connection pool size — increased from 50 to 200 in v2.14.3
    }

    upstream billing-service {
        server billing-worker:8083 max_fails=3 fail_timeout=30s;
        keepalive 50;
    }

    # Rate limiting
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=2000r/s;
    limit_conn_zone $binary_remote_addr zone=api_conn:10m;

    # SSL
    ssl_certificate /etc/ssl/certs/prod/api.company.com.crt;
    ssl_certificate_key /etc/ssl/private/api.company.com.key;

    # Main proxy config
    server {
        listen 443 ssl http2;
        server_name api.company.com;

        # Rate limiting
        limit_req zone=api_limit burst=500 nodelay;
        limit_conn api_conn 100;

        location /api/v2/ {
            proxy_pass http://user-service:8081;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;

            # Timeouts
            proxy_connect_timeout 10s;
            proxy_read_timeout 30s;
            proxy_send_timeout 30s;

            # Buffering
            proxy_buffering on;
            proxy_buffer_size 4k;
            proxy_buffers 8 4k;
        }

        location /health {
            proxy_pass http://localhost:8080/health;
            access_log off;
        }
    }
}
